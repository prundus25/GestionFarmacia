@startuml
title Validar y Reservar Stock

start

:Introducir ID de receta;
:Buscar receta en el sistema;

if (¿Receta encontrada?) then (no)
    :Mostrar error: receta no encontrada;
    stop
else (sí)
    if (¿Estado = PENDIENTE_VALIDAR?) then (no)
        :Mostrar error: estado incorrecto;
        stop
    endif

    ' Paso 1: medicamentos restringidos
    if (¿Contiene medicamento restringido?) then (sí)
        :Estado → PENDIENTE_AUTORIZACION;
        :Mostrar aviso;
        stop
    endif

    ' Paso 2: interacciones peligrosas
    if (¿Interacción peligrosa entre líneas?) then (sí)
        :Estado → PENDIENTE_MEDICO;
        :Mostrar par conflictivo;
        stop
    endif

    ' Paso 3: dosis excesiva
    if (¿Alguna dosis supera el máximo del medicamento?) then (sí)
        :Estado → PENDIENTE_MEDICO;
        :Mostrar línea afectada;
        stop
    endif

    ' Paso 4: verificación de stock línea a línea
    :stockInsuficiente = false;
    :hayAlternativas = false;

    repeat
        :Siguiente LineaReceta;
        if (¿calcularStockDisponible ≥ cajas requeridas?) then (no)
            :stockInsuficiente = true;
            if (¿Existen alternativas con stock disponible > 0?) then (sí)
                :hayAlternativas = true;
                :Mostrar alternativas disponibles\n"se ha informado al médico";
            else (no)
                :Crear OrdenReposicion para esa línea\n(idRecetaOrigen = receta.getId());
                :Mostrar aviso de orden generada;
            endif
        endif
    repeat while (¿Más líneas?) is (sí)

    if (¿stockInsuficiente = false?) then (sí)
        :Marcar receta con stock reservado;
        :Estado → STOCK_RESERVADO;
        :Mostrar confirmación;
    elseif (¿hayAlternativas = true?) then (sí)
        :Estado → PENDIENTE_MEDICO;
        :Mostrar aviso de alternativas;
    else (sin stock ni alternativas)
        :Estado → PENDIENTE_STOCK;
        :Órdenes generadas con idRecetaOrigen = receta.getId();
        :Avanzará automáticamente a STOCK_RESERVADO
al confirmar la recepción del pedido;
    endif

endif

stop
@enduml

@startuml
title Dispensar Receta

start

:Introducir ID de receta;
:Buscar receta en el sistema;

if (¿Receta encontrada?) then (no)
    :Mostrar error: receta no encontrada;
    stop
else (sí)
    if (¿Estado = STOCK_RESERVADO?) then (no)
        :Mostrar error: la receta no está\nen estado STOCK_RESERVADO;
        stop
    endif

    :Descontar stock real de cada medicamento\n(m.stock -= linea.cajas);

    fork
        :Estado → DISPENSADA;
        :Registrar fechaDispensacion;
        if (¿Receta crónica?) then (sí)
            :Inicializar seguimiento crónico\npor línea: cajasRestantes,\nduracionCaja, fechaProximaReposicion;
        endif
    fork again
        if (¿Stock ≤ stockMínimo en algún medicamento?) then (sí)
            if (¿Ya existe alerta STOCK_MINIMO activa?) then (no)
                :Crear Alerta STOCK_MINIMO;
                :Mostrar aviso;
            endif
        endif
    end fork

    :Mostrar confirmación de dispensación;
endif

stop
@enduml
